function roundedRect(x,y,w,h,r){

    ctx.beginPath();

    ctx.roundRect(
        x,y,w,h,r
    );

}


function cellRect(row,col,size){

    const cell=size/15;

    return {
        x:col*cell,
        y:row*cell,
        w:cell,
        h:cell
    };

}

function drawHome(color,row,col,size){

    const cell=size/15;

    const x=col*cell;
    const y=row*cell;

    const w=6*cell;

    const data=COLORS[color];

    const gradient=ctx.createLinearGradient(
        x,y,x+w,y+w
    );

    gradient.addColorStop(
        0,
        data.light
    );

    gradient.addColorStop(
        .35,
        data.main
    );

    gradient.addColorStop(
        1,
        data.dark
    );

    ctx.fillStyle=gradient;

    roundedRect(
        x+2,
        y+2,
        w-4,
        w-4,
        cell*.35
    );

    ctx.fill();


    /* inner white platform */

    ctx.fillStyle="#f8faff";

    roundedRect(
        x+cell*.8,
        y+cell*.8,
        w-cell*1.6,
        w-cell*1.6,
        cell*.3
    );

    ctx.fill();


    /* glossy highlight */

    const shine=ctx.createRadialGradient(
        x+w*.25,
        y+w*.2,
        0,
        x+w*.25,
        y+w*.2,
        w*.8
    );

    shine.addColorStop(
        0,
        "#ffffff66"
    );

    shine.addColorStop(
        1,
        "#ffffff00"
    );

    ctx.fillStyle=shine;

    roundedRect(
        x+2,
        y+2,
        w-4,
        w-4,
        cell*.35
    );

    ctx.fill();


    /* four token bases */

    homes[color].forEach(([rr,cc])=>{

/* =====================================================
   TOKEN LOCATION
   ===================================================== */

function getTokenPosition(
    color,
    position,
    tokenIndex
){

    if(position===-1){

        return homes[color][tokenIndex];

    }

    if(position>=52){

        return lanes[color][
            position-52
        ];

    }

    if(position===57){

        return [7,7];

    }

    const absolute=
        (starts[color]+position)%52;

    return track[absolute];

}


/* =====================================================
   DRAW TOKEN
   ===================================================== */

function drawToken(
    color,
    position,
    index,
    size
){

    const [row,col]=getTokenPosition(
        color,
        position,
        index
    );

    const cell=size/15;

    let x=col*cell+cell/2;
    let y=row*cell+cell/2;

    /*
       Slight separation when multiple
       tokens are on same cell.
    */

    if(position>=0 && position<=51){

        const same=[];

        order.slice(0,playerCount)
        .forEach(c=>{

            players[c].forEach((p,i)=>{

                if(
                    p===position &&
                    c===color
                ){

                    same.push(i);

                }

            });

        });

        if(same.length>1){

            const offset=[
                [-.22,-.22],
                [.22,-.22],
                [-.22,.22],
                [.22,.22]
            ];

            const oi=same.indexOf(index);

            if(offset[oi]){

                x+=offset[oi][0]*cell;
                y+=offset[oi][1]*cell;

            }

        }

    }


    const data=COLORS[color];

    const radius=cell*.37;


    /*
       glow
    */

    ctx.save();

    ctx.shadowColor=data.glow;

    ctx.shadowBlur=15;


    /*
       outer
    */

    const gradient=ctx.createRadialGradient(
        x-radius*.35,
        y-radius*.4,
        radius*.1,
        x,
        y,
        radius
    );

    gradient.addColorStop(
        0,
        "#ffffff"
    );

    gradient.addColorStop(
        .18,
        data.light
    );

    gradient.addColorStop(
        .55,
        data.main
    );

    gradient.addColorStop(
        1,
        data.dark
    );

    ctx.fillStyle=gradient;

    ctx.beginPath();

    ctx.arc(
        x,
        y,
        radius,
        0,
        Math.PI*2
    );

    ctx.fill();


    ctx.shadowBlur=0;

    ctx.lineWidth=cell*.08;

    ctx.strokeStyle="#ffffff";

    ctx.stroke();


    /*
       highlight
    */

    ctx.fillStyle="#ffffff88";

    ctx.beginPath();

    ctx.arc(
        x-radius*.3,
        y-radius*.35,
        radius*.2,
        0,
        Math.PI*2
    );

    ctx.fill();


    /*
       number
    */

    ctx.fillStyle="#ffffff";

    ctx.font=`bold ${cell*.30}px Arial`;

    ctx.textAlign="center";
    ctx.textBaseline="middle";

    ctx.fillText(
        index+1,
        x,
        y+cell*.05
    );


    /*
       legal move ring
    */

    if(
        color===order[currentPlayer] &&
        diceRolled &&
        canMove(color,index)
    ){

        ctx.strokeStyle="#00f0ff";

        ctx.lineWidth=cell*.09;

        ctx.shadowColor="#00eaff";

        ctx.shadowBlur=18;

        ctx.beginPath();

        ctx.arc(
            x,
            y,
            radius+cell*.12,
            0,
            Math.PI*2
        );

        ctx.stroke();

    }

    ctx.restore();

}


/* =====================================================
   DRAW EVERYTHING
   ===================================================== */

function draw(){

    const size=parseFloat(
        canvas.style.width
    ) || 600;

    ctx.clearRect(
        0,0,size,size
    );

    drawBoardBackground(size);

    /*
       outer glass frame
    */

    ctx.fillStyle="#111827";

    roundedRect(
        0,
        0,
        size,
        size,
        24
    );

    ctx.fill();


    drawHome(
        "red",
        0,
        0,
        size
    );

    drawHome(
        "green",
        0,
        9,
        size
    );

    drawHome(
        "blue",
        9,
        0,
        size
    );

    drawHome(
        "yellow",
        9,
        9,
        size
    );


    drawTrack(size);

    drawLanes(size);

    drawCenter(size);


    /*
       tokens
    */

    order.slice(0,playerCount)
    .forEach(color=>{

        players[color].forEach(
            (position,index)=>{

                if(position!==57){

                    drawToken(
                        color,
                        position,
                        index,
                        size
                    );

                }

            }
        );

    });


    /*
       finished tokens in center
    */

    drawFinishedTokens(size);

}


/* =====================================================
   FINISHED TOKENS
   ===================================================== */

function drawFinishedTokens(size){

    const cell=size/15;

    const finished=[];

    order.slice(0,playerCount)
    .forEach(color=>{

        players[color].forEach(p=>{

            if(p===57){

                finished.push(color);

            }

        });

    });

    finished.forEach((color,i)=>{

        const angle=
            (Math.PI*2/Math.max(
                finished.length,
                1
            ))*i;

        const radius=cell*.65;

        const x=
            7.5*cell+
            Math.cos(angle)*radius;

        const y=
            7.5*cell+
            Math.sin(angle)*radius;

        const data=COLORS[color];

        ctx.beginPath();

        ctx.arc(
            x,
            y,
            cell*.25,
            0,
            Math.PI*2
        );

        ctx.fillStyle=data.main;

        ctx.shadowColor=data.glow;

        ctx.shadowBlur=12;

        ctx.fill();

        ctx.shadowBlur=0;

    });

}


/* =====================================================
   LEGAL MOVE
   ===================================================== */

function canMove(color,index){

    if(!diceRolled)return false;

    const p=players[color][index];

    /*
       finished
    */

    if(p===57){
        return false;
    }

    /*
       base
    */

    if(p===-1){

        return diceValue===6;

    }

    /*
       normal
    */

    return p+diceValue<=57;

}


/* =====================================================
   GET LEGAL TOKENS
   ===================================================== */

function legalTokens(){

    const color=order[currentPlayer];

    const result=[];

    players[color].forEach(
        (p,index)=>{

            if(
                p!==57 &&
                canMove(color,index)
            ){

                result.push(index);

            }

        }
    );

    return result;

}


/* =====================================================
   DICE
   ===================================================== */

const diceFaces=[
    "",
    "⚀",
    "⚁",
    "⚂",
    "⚃",
    "⚄",
    "⚅"
];


function rollDice(){

    if(gameOver)return;

    if(diceRolled)return;

    diceValue=
        Math.floor(Math.random()*6)+1;

    diceRolled=true;

    diceElement.textContent=
        diceFaces[diceValue];


    const legal=legalTokens();

    if(legal.length===0){

        messageText.textContent=
            "No legal move. Turn lost.";

        setTimeout(
            nextTurn,
            900
        );

        return;

    }


    messageText.textContent=
        `🎲 ${COLORS[order[currentPlayer]].name} rolled ${diceValue}. Select a glowing token.`;

    draw();

}


/* =====================================================
   MOVE TOKEN
   ===================================================== */

function moveToken(index){

    if(gameOver)return;

    const color=
        order[currentPlayer];

    if(!canMove(color,index)){
        return;
    }

    let position=
        players[color][index];


    /*
       leave base
    */

    if(position===-1){

        position=0;

    }else{

        position+=diceValue;

    }


    players[color][index]=position;


    /*
       capture
    */

    if(
        position>=0 &&
        position<=51
    ){

        captureOpponents(
            color,
            position
        );

    }


    /*
       winner
    */

    if(
        players[color].every(
            p=>p===57
        )
    ){

        gameOver=true;

        clearInterval(timerInterval);

        turnText.textContent=
            `🏆 ${COLORS[color].emoji} ${COLORS[color].name} WINS!`;

        messageText.textContent=
            `Congratulations! ${COLORS[color].name} completed all 4 tokens.`;

        rollBtn.disabled=true;

        draw();

        return;

    }


    /*
       Six = extra turn
    */

    if(diceValue===6){

        consecutiveSixes++;

        if(consecutiveSixes>=3){

            messageText.textContent=
                "Three 6s! Turn lost.";

            setTimeout(
                nextTurn,
                800
            );

            return;

        }

        diceRolled=false;

        messageText.textContent=
            "🔥 You rolled 6! Roll again.";

        resetTimer();

    }else{

        nextTurn();

    }


    draw();

}


/* =====================================================
   CAPTURE
   ===================================================== */

function captureOpponents(
    color,
    position
){

    /*
       safe cells cannot capture
    */

    const absolute=
        (
            starts[color]+position
        )%52;

    if(
        safeCells.includes(absolute)
    ){

        return;

    }


    order.slice(0,playerCount)
    .forEach(other=>{

        if(other===color)return;


        players[other].forEach(
            (otherPosition,index)=>{

                if(
                    otherPosition<0 ||
                    otherPosition>51
                ){

                    return;

                }


                const otherAbsolute=
                    (
                        starts[other]+
                        otherPosition
                    )%52;


                if(
                    otherAbsolute===
                    absolute
                ){

                    players[other][index]=-1;

                    messageText.textContent=
                        `💥 ${COLORS[color].name} captured ${COLORS[other].name}!`;

                }

            }
        );

    });

}


/* =====================================================
   NEXT TURN
   ===================================================== */

function nextTurn(){

    if(gameOver)return;

    currentPlayer++;

    if(
        currentPlayer>=playerCount
    ){

        currentPlayer=0;

    }


    diceValue=0;

    diceRolled=false;

    consecutiveSixes=0;

    diceElement.textContent="⚀";


    const color=
        order[currentPlayer];

    turnText.textContent=
        `${COLORS[color].emoji} ${COLORS[color].name}'s Turn`;

    messageText.textContent=
        "Roll the dice.";

    resetTimer();

    draw();

}


/* =====================================================
   TIMER
   ===================================================== */

function resetTimer(){

    clearInterval(timerInterval);

    timer=15;

    timerText.textContent=timer;


    timerInterval=setInterval(()=>{

        timer--;

        timerText.textContent=timer;


        if(timer<=0){

            clearInterval(timerInterval);

            messageText.textContent=
                "⏰ Time Up! Turn lost.";

            nextTurn();

        }

    },1000);

}


/* =====================================================
   NEW GAME
   ===================================================== */

function newGame(){

    clearInterval(timerInterval);

    gameOver=false;

    currentPlayer=0;

    diceValue=0;

    diceRolled=false;

    consecutiveSixes=0;

    diceElement.textContent="⚀";

    rollBtn.disabled=false;

    resetPlayers();

    turnText.textContent=
        "🔴 Red's Turn";

    messageText.textContent=
        "Roll the dice.";

    resizeCanvas();

    resetTimer();

}

/* =====================================================
   CANVAS CLICK
   ===================================================== */

canvas.addEventListener(
    "click",
    function(event){

        if(gameOver)return;

        if(!diceRolled)return;

        const rect=
            canvas.getBoundingClientRect();

        const size=
            parseFloat(canvas.style.width);

        const x=
            (event.clientX-rect.left)*
            size/rect.width;

        const y=
            (event.clientY-rect.top)*
            size/rect.height;

        const cell=size/15;

        const color=
            order[currentPlayer];

        const legal=legalTokens();


        for(const index of legal){

            const [row,col]=
                getTokenPosition(
                    color,
                    players[color][index],
                    index
                );

            const tx=
                col*cell+
                cell/2;

            const ty=
                row*cell+
                cell/2;

            const distance=
                Math.hypot(
                    x-tx,
                    y-ty
                );


            if(
                distance<
                cell*.55
            ){

                moveToken(index);

                return;

            }

        }

    }
);
